use std::{fs::File, io::Write};

const FLASH_SCORE: u64 = 1_100_000_000;
const TOP_SCORE: u64 = 1_000_000_000;
const ZONE_SCORE: u64 = TOP_SCORE / 2;
const NUM_QUALIFIER_BOULDERS: usize = 30;

/// The offset into the csv where the boulder scores start
const BOULDER_START: usize = 5;
const FIRST_NAME: usize = 1;
const LAST_NAME: usize = 2;
const CATEGORY: usize = 3;

// Change to true for gender specific scores
const SEPERATE_CATEGORIES: bool = true;

fn main() -> Result<(), Error> {
    let mut args = std::env::args().skip(1);

    // Read in the path for the responses from the form
    let qualifer_csv = args
        .next()
        .map(|path| std::fs::read_to_string(path))
        .expect("Please provide the path to the comp csv")?;

    let (mut competitors, boulders) = parse_csv(qualifer_csv)?;

    // calculate scores
    for competitor in &mut competitors {
        for (outcome, boulder) in competitor.outcomes.into_iter().zip(&boulders) {
            match outcome {
                Some(Outcome::Zone) => {
                    competitor.category_specific_score +=
                        boulder.split_zone_score(competitor.category);
                    competitor.score += boulder.zone_score()
                }
                Some(Outcome::Top) => {
                    competitor.category_specific_score +=
                        boulder.split_top_score(competitor.category);
                    competitor.score += boulder.top_score()
                }
                Some(Outcome::Flash) => {
                    competitor.category_specific_score +=
                        boulder.split_flash_score(competitor.category);
                    competitor.score += boulder.flash_score()
                }
                None => {}
            }
        }
    }

    // Sort largest to smallest
    competitors.sort_by_key(Competitor::inverse_score);

    // compute spacing for nice formatting
    let spacing = 1 + competitors
        .iter()
        .map(|c| c.first_name.chars().count() + c.last_name.chars().count())
        .max()
        .unwrap_or_default();

    // mens
    {
        let mut mens = File::create("mens.md")?;
        mens.write_all(b"|Position|Name|Score|\n|--------|----|-----|\n")?;
        for (
            Competitor {
                first_name,
                last_name,
                score,
                category_specific_score,
                ..
            },
            pos,
        ) in competitors
            .iter()
            .filter(|Competitor { category, .. }| matches!(*category, Category::Mens))
            .zip(1..)
        {
            let mut buf = format!("|{pos:2}|{first_name} {last_name}|");
            for _ in 0..spacing - first_name.chars().count() - last_name.chars().count() {
                buf.push(' ')
            }
            let score = if SEPERATE_CATEGORIES {
                *category_specific_score
            } else {
                *score
            } as f64
                * 0.000001;
            buf.push_str(&format!("{score:8.2}|\n"));
            mens.write_all(buf.as_bytes())?
        }
    }

    // womens
    {
        let mut womens = File::create("womens.md")?;
        womens.write_all(b"|Position|Name|Score|\n|--------|----|-----|\n")?;
        for (
            Competitor {
                first_name,
                last_name,
                score,
                category_specific_score,
                ..
            },
            pos,
        ) in competitors
            .iter()
            .filter(|Competitor { category, .. }| matches!(*category, Category::Womens))
            .zip(1..)
        {
            let mut buf = format!("|{pos:2}|{first_name} {last_name}|");
            for _ in 0..spacing - first_name.chars().count() - last_name.chars().count() {
                buf.push(' ')
            }
            let score = if SEPERATE_CATEGORIES {
                *category_specific_score
            } else {
                *score
            } as f64
                * 0.000001;
            buf.push_str(&format!("{score:8.2}|\n"));
            womens.write_all(buf.as_bytes())?
        }
    }

    // boulders
    {
        let mut boulders_file = File::create("boulders.md")?;

        boulders_file.write_all(
            format!(
                "| Boulder    | {:11} | {:11} | {:11} | {:11} | {:11} | {:11}|\n|------------|-------------|-------------|-------------|-------------|-------------|------------|\n",
                "zone score", "num zones", "top score", "num tops", "flash score", "num flashes"
            )
            .as_bytes(),
        )?;

        for (boulder, num) in boulders.into_iter().zip(1..) {
            boulders_file.write_all(
                format!(
                    "| Boulder {num:2} | {:11.2} | {:11.2} | {:11.2} | {:11.2} | {:11.2} | {:11.2}|\n",
                    boulder.zone_score() as f64 * 0.000001,
                    boulder.zones(),
                    boulder.top_score() as f64 * 0.000001,
                    boulder.tops(),
                    boulder.flash_score() as f64 * 0.000001,
                    boulder.flashes(),
                )
                .as_bytes(),
            )?
        }
    }

    Ok(())
}

fn parse_csv(
    qualifier_csv: String,
) -> Result<(Vec<Competitor>, [Boulder; NUM_QUALIFIER_BOULDERS]), Error> {
    let mut competitors = Vec::<Competitor>::new();
    let mut boulders = [const { Boulder::new() }; NUM_QUALIFIER_BOULDERS];

    let trim = |c: char| -> bool { c == '"' || c.is_whitespace() };

    for form_entry in qualifier_csv.lines().skip(1) {
        let columns = form_entry.split(',').collect::<Vec<_>>();

        let first_name = columns[FIRST_NAME].trim_matches(trim).to_string();
        let last_name = columns[LAST_NAME].trim_matches(trim).to_string();

        // Check for duplicates
        if competitors.iter().any(|competitor| {
            competitor.first_name.to_lowercase() == first_name.to_lowercase()
                && competitor.last_name.to_lowercase() == last_name.to_lowercase()
        }) {
            continue;
        }

        let category = Category::parse(columns[CATEGORY])?;

        let mut outcomes = [const { None }; NUM_QUALIFIER_BOULDERS];
        for (i, boulder) in boulders.iter_mut().enumerate() {
            let outcome = Outcome::parse(columns[i + BOULDER_START]).ok();

            use {Category as C, Outcome as O};
            outcomes[i] = match (outcome, category) {
                // mens
                (Some(O::Zone), C::Mens) => {
                    boulder.men_zones += 1;
                    Some(Outcome::Zone)
                }
                (Some(O::Top), C::Mens) => {
                    boulder.men_tops += 1;
                    Some(Outcome::Top)
                }
                (Some(O::Flash), C::Mens) => {
                    boulder.men_flashes += 1;
                    Some(Outcome::Flash)
                }

                // womens
                (Some(O::Zone), C::Womens) => {
                    boulder.women_zones += 1;
                    Some(Outcome::Zone)
                }
                (Some(O::Top), C::Womens) => {
                    boulder.women_tops += 1;
                    Some(Outcome::Top)
                }
                (Some(O::Flash), C::Womens) => {
                    boulder.women_flashes += 1;
                    Some(Outcome::Flash)
                }

                (None, _) => None,
            }
        }

        competitors.push(Competitor {
            first_name,
            last_name,
            category,
            score: 0,                   // Calculate this later
            category_specific_score: 0, // Calculate this later
            outcomes,
        })
    }

    Ok((competitors, boulders))
}

#[derive(Clone, Copy)]
struct Boulder {
    men_zones: u64,
    men_tops: u64,
    men_flashes: u64,
    women_zones: u64,
    women_tops: u64,
    women_flashes: u64,
}

impl Boulder {
    const fn new() -> Self {
        Self {
            men_zones: 0,
            men_tops: 0,
            men_flashes: 0,
            women_zones: 0,
            women_tops: 0,
            women_flashes: 0,
        }
    }

    const fn zones(&self) -> u64 {
        self.women_zones + self.men_zones
    }
    const fn tops(&self) -> u64 {
        self.women_tops + self.men_tops
    }
    const fn flashes(&self) -> u64 {
        self.women_flashes + self.men_flashes
    }

    const fn zone_score(&self) -> u64 {
        let total = self.zones() + self.tops() + self.flashes();
        match total {
            0..=1 => ZONE_SCORE,
            t => ZONE_SCORE / t,
        }
    }

    const fn top_score(&self) -> u64 {
        let total = self.men_tops + self.men_flashes + self.women_tops + self.women_flashes;

        match total {
            0..=1 => TOP_SCORE,
            t => TOP_SCORE / t,
        }
    }

    const fn flash_score(&self) -> u64 {
        let total = self.men_tops + self.men_flashes + self.women_tops + self.women_flashes;

        match total {
            0..=1 => FLASH_SCORE,
            t => FLASH_SCORE / t,
        }
    }

    const fn split_zone_score(&self, category: Category) -> u64 {
        let total = match category {
            Category::Mens => self.men_zones + self.men_tops + self.men_flashes,
            Category::Womens => self.women_zones + self.women_tops + self.women_flashes,
        };
        match total {
            0..=1 => ZONE_SCORE,
            t => ZONE_SCORE / t,
        }
    }

    const fn split_top_score(&self, category: Category) -> u64 {
        let total = match category {
            Category::Mens => self.men_tops + self.men_flashes,
            Category::Womens => self.women_tops + self.women_flashes,
        };

        match total {
            0..=1 => TOP_SCORE,
            t => TOP_SCORE / t,
        }
    }

    const fn split_flash_score(&self, category: Category) -> u64 {
        let total = match category {
            Category::Mens => self.men_tops + self.men_flashes,
            Category::Womens => self.women_tops + self.women_flashes,
        };

        match total {
            0..=1 => FLASH_SCORE,
            t => FLASH_SCORE / t,
        }
    }
}

#[derive(Debug)]
struct Competitor {
    first_name: String,
    last_name: String,
    category: Category,
    score: u64,
    category_specific_score: u64,
    outcomes: [Option<Outcome>; NUM_QUALIFIER_BOULDERS],
}

#[derive(Debug, Clone, Copy)]
enum Outcome {
    Zone,
    Top,
    Flash,
}

#[derive(Debug, Clone, Copy)]
enum Category {
    Mens,
    Womens,
}

#[derive(Debug)]
pub enum Error {
    CategoryError(String),
    OutcomeError(String),
    Io(std::io::Error),
}

impl From<std::io::Error> for Error {
    fn from(value: std::io::Error) -> Self {
        Self::Io(value)
    }
}

impl Outcome {
    fn parse(s: &str) -> Result<Self, Error> {
        match s.trim_matches(|c: char| -> bool { c == '"' || c.is_whitespace() }) {
            "Zone" => Ok(Self::Zone),
            "Top" => Ok(Self::Top),
            "Flash" => Ok(Self::Flash),
            o => Err(Error::OutcomeError(o.to_string())),
        }
    }
}

impl Category {
    fn parse(s: &str) -> Result<Self, Error> {
        match s.trim_matches(|c: char| -> bool { c == '"' || c.is_whitespace() }) {
            "Womens" => Ok(Self::Womens),
            "Mens" => Ok(Self::Mens),
            o => Err(Error::CategoryError(o.to_string())),
        }
    }
}
impl Competitor {
    fn inverse_score(&self) -> u64 {
        if SEPERATE_CATEGORIES {
            u64::MAX - self.category_specific_score
        } else {
            u64::MAX - self.score
        }
    }
}
